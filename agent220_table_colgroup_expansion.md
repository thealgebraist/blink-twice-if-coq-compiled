### 1. Span Expansion Mechanics (`<colgroup>` vs `<col>`)

In HTML5 §4.9.3 and Chromium Blink’s `LayoutTableCol` / TableNG (`NGTableLayoutAlgorithm`), column elements (`<colgroup>` and `<col>`) define column properties across a table grid without containing structural cell data.

1. **`<colgroup>` without `<col>` children:** 
   The `<colgroup>` itself represents a span of columns. Its `span` attribute (clamped to $[1, 1000]$ per HTML5 specification, defaulted to $1$) dictates how many consecutive layout columns it generates. All generated columns share the `<colgroup>`'s computed style (e.g., `width`, `background`, `visibility`).

2. **`<colgroup>` with `<col>` children:** 
   If any `<col>` element is a child of the `<colgroup>`, the `<colgroup>`'s own `span` attribute is ignored. Instead, the sequence of `<col>` children is expanded in tree order. Each `<col span=N>` expands into $N$ consecutive layout columns. 

3. **Style and Width Inheritance:**
   During expansion of `<col>` elements inside a `<colgroup>`, each generated column inherits undeclared properties from the parent `<colgroup>`'s style. Specifically for width resolution in Blink (`NGTableTypes::CreateColumnFromStyle`), if a `<col>` has an `auto` (unspecified) width, the layout column inherits the resolved width from its parent `<colgroup>`. If the `<col>` explicitly declares a fixed or percentage width, that overrides the `<colgroup>` width for all $N$ columns expanded from that `<col>`.

---

### 2. Core Verification Invariants

To verify Blink's column expansion semantics against HTML5 §4.9.3, we enforce two fundamental invariants:

* **Invariant 1 (Count Conservation):** For any `<colgroup>` containing a non-empty list of `<col>` elements, the total number of layout columns generated by the expansion algorithm is strictly equal to the sum of the clamped `span` attributes of its child `<col>` elements. If the `<col>` list is empty, the total count equals the `<colgroup>`'s own clamped `span`.
* **Invariant 2 (Width Cascade Preservation):** For every expanded layout column originating from a `<col>` element with an unspecified (`None`) width, its effective layout width must strictly equal the parent `<colgroup>`'s declared width. Furthermore, all $N$ consecutive columns generated by a single `<col span=N>` must possess identical resolved widths.

### 3. Concrete Rocq/Coq Formalization

Below is the complete, self-contained formalization for `BlinkTableColgroupExpansionSemantics.v` using standard Coq library constructs, with **0 admits**.

```coq
(** * BlinkTableColgroupExpansionSemantics.v
    Formal Audit of Chromium Blink <colgroup> and <col> Span Expansion
    against HTML5 §4.9.3. *)

Require Import Coq.Lists.List.
Require Import Coq.Arith.Arith.
Require Import Coq.Arith.PeanoNat.
Import ListNotations.

(** ** 1. Data Definitions *)

(** Width representation: None represents CSS 'auto' / unspecified width,
    Some w represents an explicit CSS length/percentage. *)
Definition StyleWidth := option nat.

(** DOM <col> element representation.
    [col_span] is the clamped span attribute (HTML5 requires span >= 1). *)
Record ColElement := mkCol {
  col_span : nat;
  col_width : StyleWidth
}.

(** DOM <colgroup> element representation. *)
Record ColGroupElement := mkColGroup {
  cg_span : nat;
  cg_width : StyleWidth;
  cg_children : list ColElement
}.

(** Expanded Layout Column in Blink's Table Grid (e.g., NGTableTypes::Column). *)
Record LayoutColumn := mkLayoutCol {
  lcol_width : StyleWidth
}.

(** ** 2. Expansion Semantics *)

(** Resolve width of a <col> against its parent <colgroup> width. *)
Definition resolve_width (cg_w col_w : StyleWidth) : StyleWidth :=
  match col_w with
  | Some w => Some w
  | None => cg_w
  end.

(** Expand a single <col> element into N consecutive LayoutColumns. *)
Definition expand_col (cg_w : StyleWidth) (c : ColElement) : list LayoutColumn :=
  let effective_w := resolve_width cg_w (col_width c) in
  repeat (mkLayoutCol effective_w) (col_span c).

(** Expand a list of <col> children inside a <colgroup>. *)
Fixpoint expand_col_list (cg_w : StyleWidth) (cols : list ColElement) : list LayoutColumn :=
  match cols with
  | [] => []
  | c :: rest => expand_col cg_w c ++ expand_col_list cg_w rest
  end.

(** Top-level <colgroup> expansion per HTML5 §4.9.3:
    If the colgroup has <col> children, its own span is ignored and children are expanded.
    Otherwise, the colgroup itself expands according to its own span and width. *)
Definition expand_colgroup (cg : ColGroupElement) : list LayoutColumn :=
  match cg_children cg with
  | [] => repeat (mkLayoutCol (cg_width cg)) (cg_span cg)
  | cols => expand_col_list (cg_width cg) cols
  end.

(** Helper: Sum of spans for a list of <col> elements. *)
Fixpoint sum_col_spans (cols : list ColElement) : nat :=
  match cols with
  | [] => 0
  | c :: rest => col_span c + sum_col_spans rest
  end.

(** ** 3. Formal Proofs of Invariants *)

(** Lemma: Expanding a single <col> produces exactly (col_span c) columns. *)
Lemma expand_col_length : forall (cg_w : StyleWidth) (c : ColElement),
  length (expand_col cg_w c) = col_span c.
Proof.
  intros cg_w c.
  unfold expand_col.
  apply repeat_length.
Qed.

(** Theorem 1A (Count Invariant for <col> list):
    The number of layout columns generated by a list of <col> elements
    equals the sum of their individual spans. *)
Theorem expand_col_list_count_invariant : forall (cg_w : StyleWidth) (cols : list ColElement),
  length (expand_col_list cg_w cols) = sum_col_spans cols.
Proof.
  intros cg_w cols.
  induction cols as [| c rest IH].
  - simpl. reflexivity.
  - simpl.
    rewrite app_length.
    rewrite expand_col_length.
    rewrite IH.
    reflexivity.
Qed.

(** Theorem 1B (Master Count Invariant for <colgroup>):
    A <colgroup> generates (cg_span) columns if childless, or (sum_col_spans) if populated. *)
Theorem expand_colgroup_count_invariant : forall (cg : ColGroupElement),
  length (expand_colgroup cg) =
    match cg_children cg with
    | [] => cg_span cg
    | cols => sum_col_spans cols
    end.
Proof.
  intros cg.
  unfold expand_colgroup.
  destruct (cg_children cg) as [| c rest] eqn:Hchildren.
  - apply repeat_length.
  - apply expand_col_list_count_invariant.
Qed.

(** Theorem 2A (Uniformity of <col> Span Expansion):
    Every column generated by a single <col span=N> shares the exact same resolved width. *)
Theorem expand_col_uniform_style : forall (cg_w : StyleWidth) (c : ColElement) (lc : LayoutColumn),
  In lc (expand_col cg_w c) ->
  lcol_width lc = resolve_width cg_w (col_width c).
Proof.
  intros cg_w c lc HIn.
  unfold expand_col in HIn.
  apply repeat_spec in HIn.
  subst lc.
  simpl. reflexivity.
Qed.

(** Theorem 2B (Width Inheritance Invariant):
    If a <col> does not specify a width (col_width = None), every layout column
    it generates inherits the parent <colgroup>'s width. *)
Theorem expand_col_inherits_colgroup_width : forall (cg_w : StyleWidth) (c : ColElement) (lc : LayoutColumn),
  col_width c = None ->
  In lc (expand_col cg_w c) ->
  lcol_width lc = cg_w.
Proof.
  intros cg_w c lc HNone HIn.
  pose proof (expand_col_uniform_style cg_w c lc HIn) as HWidth.
  rewrite HNone in HWidth.
  simpl in HWidth.
  exact HWidth.
Qed.

(** Theorem 2C (Global Colgroup Inheritance Invariant):
    If all <col> children in a <colgroup> have unspecified widths, then EVERY layout
    column generated by the <colgroup> has the <colgroup>'s width. *)
Theorem expand_colgroup_global_inheritance : forall (cg : ColGroupElement) (lc : LayoutColumn),
  Forall (fun c => col_width c = None) (cg_children cg) ->
  In lc (expand_colgroup cg) ->
  lcol_width lc = cg_width cg.
Proof.
  intros cg lc HForall HIn.
  unfold expand_colgroup in HIn.
  destruct (cg_children cg) as [| c0 rest] eqn:Hchildren.
  - apply repeat_spec in HIn.
    subst lc. simpl. reflexivity.
  - remember (c0 :: rest) as cols.
    clear Hchildren c0 rest Heqcols.
    induction cols as [| c rest IH].
    + simpl in HIn. contradiction.
    + simpl in HIn.
      apply in_app_or in HIn.
      inversion HForall as [| ? ? Hc_none Hrest_none]; subst.
      destruct HIn as [HIn_c | HIn_rest].
      * eapply expand_col_inherits_colgroup_width; eauto.
      * apply IH; auto.
Qed.
```
